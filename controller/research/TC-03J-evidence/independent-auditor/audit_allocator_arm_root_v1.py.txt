def raw_tables(path):
    raw=path.read_bytes();h=struct.unpack_from('<16sHHIIIIIHHHHHH',raw)
    sections=[struct.unpack_from('<IIIIIIIIII',raw,h[6]+i*h[11]) for i in range(h[12])]
    def data(s):return raw[s[4]:s[4]+s[5]]
    names=data(sections[h[13]])
    def string(table,n):return table[n:].split(b'\0',1)[0].decode('ascii')
    names=[string(names,s[0]) for s in sections]
    symbols={}
    for index,s in enumerate(sections):
        if s[1]!=2:continue
        table=data(sections[s[6]]);items=[]
        for offset in range(0,s[5],16):
            n,v,z,info,other,owner=struct.unpack_from('<IIIBBH',data(s),offset)
            items.append({'name':string(table,n),'value':v,'bytes':z,'binding':info>>4,
                          'type':info&15,'section':owner})
        symbols[index]=items
    relocations=[]
    for index,s in enumerate(sections):
        if s[1]!=9:continue
        assert s[9]==8 and s[5]%8==0
        for offset in range(0,s[5],8):
            position,info=struct.unpack_from('<II',data(s),offset)
            relocations.append({'section':names[s[7]],'offset':position,'type':info&255,
                                'symbol':symbols[s[6]][info>>8]['name']})
    exports=[v for table in symbols.values() for v in table if v['name'] in PUBLIC]
    assert {v['name'] for v in exports}==PUBLIC and len(exports)==3
    assert all(v['binding']==1 and v['type']==2 and v['section']>0 for v in exports)
    executable={names[i]:data(s) for i,s in enumerate(sections) if s[2]&4 and s[5]}
    return exports,relocations,executable

def disassembly_bytes(path,sections,relocations):
    text=path.read_text(encoding='utf-8');assert '<unknown>' not in text.lower()
    current=None;covered={name:set() for name in sections};listed=[];instructions=0
    last_end=0;elision_start=None;elided_zero_bytes=0
    for line in text.splitlines():
        section=re.fullmatch(r'Disassembly of section (.+):',line)
        if section:
            assert elision_start is None
            current=section[1];assert current in sections;last_end=0;continue
        if line.strip()=='...':
            assert current in sections and elision_start is None
            elision_start=last_end;continue
        rel=re.fullmatch(r'\s*([0-9a-f]+):\s+(R_ARM_[A-Z0-9_]+)\s+(\S+)\s*',line)
        if rel:
            types={'R_ARM_THM_CALL':10,'R_ARM_THM_MOVW_ABS_NC':47,'R_ARM_THM_MOVT_ABS':48}
            listed.append({'section':current,'offset':int(rel[1],16),'type':types[rel[2]],'symbol':rel[3]});continue
        prefix=line.split('\t',1)[0]
        match=re.fullmatch(r'\s*([0-9a-f]+):\s+((?:[0-9a-f]{2}|[0-9a-f]{4}|[0-9a-f]{8})(?:\s+(?:[0-9a-f]{2}|[0-9a-f]{4}|[0-9a-f]{8}))*)\s*',prefix)
        if not match:continue
        assert current in sections
        offset=int(match[1],16)
        if elision_start is not None:
            assert offset>elision_start and not any(sections[current][elision_start:offset])
            covered[current].update(range(elision_start,offset));elided_zero_bytes+=offset-elision_start
            elision_start=None
        encoded=b''.join(int(word,16).to_bytes(len(word)//2,'little') for word in match[2].split())
        assert sections[current][offset:offset+len(encoded)]==encoded
        positions=set(range(offset,offset+len(encoded)))
        assert not positions.intersection(covered[current]);covered[current].update(positions);instructions+=1
        last_end=offset+len(encoded)
    assert elision_start is None
    assert all(covered[name]==set(range(len(raw))) for name,raw in sections.items())
    expected=[v for v in relocations if v['section'] in sections]
    assert sorted(listed,key=str)==sorted(expected,key=str)
    return {'listed_instruction_or_data_lines':instructions,'executable_bytes_reconciled':sum(map(len,sections.values())),
            'executable_relocations_reconciled':len(listed),'all_executable_section_bytes_covered':True,
            'objdump_explicitly_elided_zero_bytes_verified_from_raw_ELF':elided_zero_bytes}
