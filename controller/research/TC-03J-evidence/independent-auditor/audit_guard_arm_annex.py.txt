def sha(path):
    h=hashlib.sha256()
    with path.open('rb') as stream:
        for chunk in iter(lambda:stream.read(65536),b''):h.update(chunk)
    return h.hexdigest()
def read(p):return json.loads(p.read_text(encoding='utf-8'))
def elf(path):
    raw=path.read_bytes();header=struct.unpack_from('<16sHHIIIIIHHHHHH',raw)
    assert raw[:7]==b'\x7fELF\x01\x01\x01' and header[1:3]==(1,40)
    offset,stride,count,names_index=header[6],header[11],header[12],header[13]
    assert stride==40 and names_index<count and offset+count*stride<=len(raw)
    sections=[struct.unpack_from('<IIIIIIIIII',raw,offset+i*stride) for i in range(count)]
    def data(s):
        assert s[1]==8 or s[4]+s[5]<=len(raw)
        return raw[s[4]:s[4]+s[5]]
    names=data(sections[names_index])
    def name(s):return names[s[0]:].split(b'\0',1)[0].decode('ascii')
    sizes={'text':0,'data':0,'bss':0};undefined=[];attrs=None
    for s in sections:
        if s[2]&2:sizes['bss' if s[1]==8 else 'data' if s[2]&1 else 'text']+=s[5]
        if name(s)=='.ARM.attributes':attrs=data(s)
        if s[1]==2:
            assert s[9]==16 and s[5]%16==0
            strings=data(sections[s[6]])
            for pos in range(0,s[5],16):
                n,v,sz,info,other,index=struct.unpack_from('<IIIBBH',data(s),pos)
                if n and index==0:undefined.append(strings[n:].split(b'\0',1)[0].decode('ascii'))
    sizes['total']=sum(sizes.values())
    assert attrs and attrs[0]==65
    def integer(i,end):
        n,shift=0,0
        while i<end:
            b=attrs[i];i+=1;n|=(b&127)<<shift
            if b<128:return n,i
            shift+=7;assert shift<35
        raise AssertionError('truncated ULEB')
    def string(i,end):
        j=attrs.index(0,i,end);return attrs[i:j].decode('ascii'),j+1
    tags={};pos=1
    while pos<len(attrs):
        start=pos;end=start+struct.unpack_from('<I',attrs,pos)[0]
        assert start+5<=end<=len(attrs)
        vendor,pos=string(pos+4,end)
        while pos<end:
            begin=pos;kind,pos=integer(pos,end);finish=begin+struct.unpack_from('<I',attrs,pos)[0];pos+=4
            assert pos<=finish<=end
            if vendor!='aeabi' or kind!=1:pos=finish;continue
            while pos<finish:
                tag,pos=integer(pos,finish)
                if tag in (4,5) or tag>=32 and tag%2:value,pos=string(pos,finish)
                else:
                    value,pos=integer(pos,finish)
                    if tag==32:
                        text,pos=string(pos,finish);value=[value,text]
                assert tag not in tags or tags[tag]==value
                tags[tag]=value
        pos=end
    return {'flags':header[7],'sections':[{'name':name(s),'type':s[1],'flags':s[2],'bytes':s[5]} for s in sections],
            'sizes':sizes,'undefined':sorted(undefined),'tags':tags}

def dependencies(path,snapshot,toolchain):
    text=path.read_text().replace('\\\n',' ')
    assert ': ' in text
    body=text.split(': ',1)[1];items=[];word='';i=0
    while i<len(body):
        c=body[i]
        if c=='\\' and i+1<len(body) and body[i+1] in ' \t#:':word+=body[i+1];i+=2;continue
        if c.isspace():
            if word:items.append(word);word=''
        else:word+=c
        i+=1
    if word:items.append(word)
    result=[]
    for item in items:
        p=(snapshot/item).resolve()
        if p.is_relative_to(snapshot):owner,base='native_snapshot',snapshot
        else:owner,base='verified_toolchain',toolchain
        assert p.is_relative_to(base) and p.is_file()
        result.append({'owner':owner,'path':p.relative_to(base).as_posix(),'sha256':sha(p)})
    return result
