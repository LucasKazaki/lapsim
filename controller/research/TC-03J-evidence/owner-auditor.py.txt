from pathlib import Path
import hashlib
import json
import struct

W = Path(__file__).resolve().parent.parent
ROOT = W / "work/enme408-allocator-cortex-m4-v1"


def sha(path):
    h = hashlib.sha256()
    with path.open("rb") as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()


checks = []


def check(label, value):
    checks.append({"check": label, "passed": bool(value)})
    if not value:
        raise AssertionError(label)


def elf_facts(path):
    raw = path.read_bytes()
    h = struct.unpack_from("<16sHHIIIIIHHHHHH", raw)
    check(path.name + " raw ELF32 little ARM relocatable", raw[:6] == b"\x7fELF\x01\x01" and h[1:3] == (1, 40))
    offset, stride, count, names_index = h[6], h[11], h[12], h[13]
    check("bounded section table", stride == 40 and 0 < count and names_index < count and offset + stride * count <= len(raw))
    sections = [struct.unpack_from("<10I", raw, offset + i * stride) for i in range(count)]
    strings = sections[names_index]
    names = raw[strings[4]:strings[4]+strings[5]]
    def cstring(data, position):
        return data[position:data.index(0, position)].decode("ascii")
    records = []
    exports, undefined = [], []
    for index, s in enumerate(sections):
        name = cstring(names, s[0])
        check("section bounds " + name, s[1] == 8 or s[4] + s[5] <= len(raw))
        records.append({"name": name, "type": s[1], "flags": s[2], "bytes": s[5]})
        if s[1] != 2:
            continue
        check("symbol table shape", s[9] == 16 and s[5] % 16 == 0 and s[6] < count)
        string_section = sections[s[6]]
        table = raw[string_section[4]:string_section[4]+string_section[5]]
        for position in range(s[4], s[4]+s[5], 16):
            n, value, size, info, other, section = struct.unpack_from("<IIIBBH", raw, position)
            symbol = cstring(table, n)
            if section == 0 and symbol:
                undefined.append(symbol)
            if symbol in ("tc_allocator_version", "tc_allocator_abi_sizeof", "tc_allocate_propulsive_torque"):
                check("raw defined global Thumb function " + symbol, section != 0 and info >> 4 == 1 and info & 15 == 2 and value & 1 == 1)
                exports.append({"name": symbol, "bytes": size, "value": value, "section_index": section})
    check("three allocator exports", len(exports) == 3)
    return {"flags": h[7], "sections": records, "public_functions": exports, "undefined_symbols": sorted(undefined)}

    check("raw undefined symbol set equals nm", raw["undefined_symbols"] == sorted(row["undefined_symbols"]))
